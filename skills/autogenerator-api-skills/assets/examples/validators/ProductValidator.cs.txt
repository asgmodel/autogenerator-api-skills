// Illustrative adaptation: replace host namespaces and resolve dependency imports.
// Business assumptions: name required, optional HTTP(S) website, nonnegative price,
// a tag is required for publication, and publication requires an active category.
using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;
using AutoGenerator.Conditions;
using Example.Models;
using v1.Validators;
using v1.Validators.Conditions;
using WasmAI.ConditionChecker.Base;

namespace Example.Validators;

public enum ProductValidatorStates
{
    IsFound,
    HasName,
    HasValidOptionalUrl,
    MatchesActive,
    HasValidPrice,
    HasItems,
    HasKind,
    IsDigital,
    IsPhysical,
    IsInKinds,
    HasActiveCategory,
    CanPublish
}

public static class ProductFeatureKeys
{
    public const string Active = "is_active";
    public const string Kind = "kind";
}

public static class ProductKinds
{
    public const string Digital = "digital";
    public const string Physical = "physical";
}

public class ProductValidatorContext
    : ValidatorContext<Product, ProductValidatorStates>, ITValidator
{
    public ProductValidatorContext(IConditionChecker checker) : base(checker) { }
    protected override void InitializeConditions() { }

    [RegisterConditionValidator(typeof(ProductValidatorStates),
        ProductValidatorStates.IsFound, "Product not found.")]
    private Task<ConditionResult> ValidateFound(DataFilter<string, Product> f)
        => f.Share is not null
            ? ConditionResult.ToSuccessAsync(f.Share)
            : ConditionResult.ToFailureAsync("Product not found.");

    [RegisterConditionValidator(typeof(ProductValidatorStates),
        ProductValidatorStates.HasName, "Product name is required.")]
    private Task<ConditionResult> ValidateName(DataFilter<string, Product> f)
        => !string.IsNullOrWhiteSpace(f.Share?.Name)
            ? ConditionResult.ToSuccessAsync(f.Share)
            : ConditionResult.ToFailureAsync("Product name is required.");

    [RegisterConditionValidator(typeof(ProductValidatorStates),
        ProductValidatorStates.HasValidOptionalUrl, "Website must be an HTTP(S) URL.")]
    private Task<ConditionResult> ValidateUrl(DataFilter<string, Product> f)
    {
        if (f.Share is null)
            return ConditionResult.ToFailureAsync("Product not found.");
        var value = f.Share.Website;
        var valid = value is null || (Uri.TryCreate(value, UriKind.Absolute, out var uri)
            && (uri.Scheme == Uri.UriSchemeHttp || uri.Scheme == Uri.UriSchemeHttps));
        return valid ? ConditionResult.ToSuccessAsync(f.Share)
            : ConditionResult.ToFailureAsync("Website must be an HTTP(S) URL.");
    }

    [RegisterConditionValidator(typeof(ProductValidatorStates),
        ProductValidatorStates.MatchesActive, "Active state does not match.",
        Value = ProductFeatureKeys.Active)]
    private Task<ConditionResult> ValidateActive(DataFilter<bool, Product> f)
        => f.Share is not null && f.Share.IsActive == f.Value
            ? ConditionResult.ToSuccessAsync(f.Share)
            : ConditionResult.ToFailureAsync("Active state does not match.");

    [RegisterConditionValidator(typeof(ProductValidatorStates),
        ProductValidatorStates.HasValidPrice, "Price must be nonnegative.")]
    private Task<ConditionResult> ValidatePrice(DataFilter<decimal, Product> f)
        => HasNonnegativePrice(f.Share)
            ? ConditionResult.ToSuccessAsync(f.Share)
            : ConditionResult.ToFailureAsync("Price must be nonnegative.");

    private static bool HasNonnegativePrice(Product? product)
        => product is not null && product.Price >= 0m;

    [RegisterConditionValidator(typeof(ProductValidatorStates),
        ProductValidatorStates.HasItems, "At least one nonblank tag is required.")]
    private Task<ConditionResult> ValidateTags(DataFilter<string, Product> f)
        => f.Share?.Tags?.Any(t => !string.IsNullOrWhiteSpace(t)) == true
            ? ConditionResult.ToSuccessAsync(f.Share)
            : ConditionResult.ToFailureAsync("At least one nonblank tag is required.");

    [RegisterConditionValidator(typeof(ProductValidatorStates),
        ProductValidatorStates.HasKind, "Product kind does not match.",
        Value = ProductFeatureKeys.Kind)]
    private Task<ConditionResult> ValidateKind(DataFilter<string, Product> f)
        => f.Share is not null && !string.IsNullOrWhiteSpace(f.Value)
            && string.Equals(f.Share.Kind, f.Value, StringComparison.OrdinalIgnoreCase)
            ? ConditionResult.ToSuccessAsync(f.Share)
            : ConditionResult.ToFailureAsync("Product kind does not match.");

    // Fixed states deliberately have separate handlers until the host confirms
    // whether attribute Value is automatically copied to f.Value.
    [RegisterConditionValidator(typeof(ProductValidatorStates),
        ProductValidatorStates.IsDigital, "Product is not digital.")]
    private Task<ConditionResult> ValidateDigital(DataFilter<string, Product> f)
        => MatchFixedKind(f.Share, ProductKinds.Digital);

    [RegisterConditionValidator(typeof(ProductValidatorStates),
        ProductValidatorStates.IsPhysical, "Product is not physical.")]
    private Task<ConditionResult> ValidatePhysical(DataFilter<string, Product> f)
        => MatchFixedKind(f.Share, ProductKinds.Physical);

    private static Task<ConditionResult> MatchFixedKind(Product? product, string expected)
        => product is not null && string.Equals(product.Kind, expected, StringComparison.OrdinalIgnoreCase)
            ? ConditionResult.ToSuccessAsync(product)
            : ConditionResult.ToFailureAsync("Product kind does not match.");

    [RegisterConditionValidator(typeof(ProductValidatorStates),
        ProductValidatorStates.IsInKinds, "Product kind is not allowed.")]
    private Task<ConditionResult> ValidateAllowedKinds(DataFilter<List<string>, Product> f)
        => f.Share is not null && !string.IsNullOrWhiteSpace(f.Share.Kind)
            && f.Value?.Any(kind => string.Equals(kind, f.Share.Kind,
                StringComparison.OrdinalIgnoreCase)) == true
            ? ConditionResult.ToSuccessAsync(f.Share)
            : ConditionResult.ToFailureAsync("Product kind is not allowed.");

    [RegisterConditionValidator(typeof(ProductValidatorStates),
        ProductValidatorStates.HasActiveCategory, "Category is missing or inactive.")]
    private async Task<ConditionResult> ValidateCategory(DataFilter<string, Product> f)
    {
        if (f.Share is null || string.IsNullOrWhiteSpace(f.Share.CategoryId))
            return ConditionResult.ToFailure(null, "Category is missing.");

        var result = await _checker.CheckAndResultAsync(
            CategoryValidatorStates.IsActive, f.Share.CategoryId);
        if (result.Success != true)
            return result;

        return result.Result is Category
            ? ConditionResult.ToSuccess(f.Share)
            : ConditionResult.ToFailure(null, "Invalid category result.");
    }

    [RegisterConditionValidator(typeof(ProductValidatorStates),
        ProductValidatorStates.CanPublish, "Product cannot be published.")]
    private async Task<ConditionResult> ValidatePublication(DataFilter<string, Product> f)
    {
        if (f.Share is null)
            return ConditionResult.ToFailure(null, "Product not found.");

        // Reuse pure handlers rather than duplicating business rules.
        var name = await ValidateName(f);
        if (name.Success != true) return name;

        var tags = await ValidateTags(f);
        if (tags.Success != true) return tags;

        // Direct pure predicate is shared with the registered price condition.
        // Do not create an incompatible DataFilter just to reuse a handler.
        if (!HasNonnegativePrice(f.Share))
            return ConditionResult.ToFailure(null, "Price must be nonnegative.");

        var category = await ValidateCategory(f);
        if (category.Success != true) return category;

        return ConditionResult.ToSuccess(f.Share);
    }
}
